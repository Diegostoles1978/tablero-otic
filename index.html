async function load() { try { const url = "https://docs.google.com/spreadsheets/d/e/2PACX-1vQZSnc6zf2k6ICNeut_QXegHFOWtDdiSX_LY4qg1LqP4z3iYNUwcXMdgNaZODcy8Q/pub?output=csv"; const res = await fetch(url + "&t=" + new Date().getTime()); const csv = await res.text(); // ✅ dividir filas const rows = csv.split(/\r?\n/); // ✅ función correcta para dividir columnas (respeta comillas) function splitRow(row) { const result = []; let current = ""; let insideQuotes = false; for (let char of row) { if (char === '"') { insideQuotes = !insideQuotes; } else if (char === ',' && !insideQuotes) { result.push(current); current = ""; } else { current += char; } } result.push(current); return result; } // ✅ construir tabla correcta const table = rows.map(splitRow); // ✅ fila 3 real const r = table[2]; // ✅ posiciones fijas (ahora sí no se corren) const nombre = r[2]; const modalidad = r[3]; const responsable = r[6]; const estado = r[8]; // ✅ limpiar presupuesto correctamente const presupuesto = Number( (r[4] || "") .replace(/\$/g, "") .replace(/,/g, "") ); document.getElementById("card").innerHTML = `

${nombre}

👤 ${responsable}

📄 ${modalidad}

💰 $${presupuesto.toLocaleString('es-CO')}

Estado: ${estado}

`; } catch (e) { console.error(e); document.getElementById("card").innerHTML = "❌ Error cargando datos"; } }
